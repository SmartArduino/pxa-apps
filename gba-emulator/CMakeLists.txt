cmake_minimum_required(VERSION 3.20)
project(pxa_gba C CXX)
if(EMU_NATIVE_TEST)
    add_executable(gba_pacer_test ../common/emulator/tests/pacer_test.c)
    target_compile_options(gba_pacer_test PRIVATE -UNDEBUG)
    target_compile_definitions(gba_pacer_test PRIVATE EMU_PACER_VISIBLE_BUDGET_US=70000u)
    add_test(NAME gba_pacer COMMAND gba_pacer_test)
    set(CMAKE_CXX_STANDARD 17)
else()
    set(CMAKE_CXX_STANDARD 26)
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/engine.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/../common/emulator/BuildEmulator.cmake")
if(EMU_NATIVE_TEST)
    add_executable(gba_rom_probe tests/rom_probe.c)
    target_link_libraries(gba_rom_probe PRIVATE emu_core)
    target_compile_options(gba_rom_probe PRIVATE -UNDEBUG)
    add_executable(gba_paging_test tests/paging_test.c)
    target_link_libraries(gba_paging_test PRIVATE emu_core)
    target_compile_options(gba_paging_test PRIVATE -UNDEBUG)
    add_test(NAME gba_paging COMMAND gba_paging_test "${CMAKE_CURRENT_BINARY_DIR}/test.rom")
    add_executable(gba_cpu_flags_test tests/cpu_flags_test.c)
    target_link_libraries(gba_cpu_flags_test PRIVATE emu_core)
    target_compile_options(gba_cpu_flags_test PRIVATE -UNDEBUG)
    add_test(NAME gba_cpu_flags COMMAND gba_cpu_flags_test "${CMAKE_CURRENT_BINARY_DIR}/test.rom")
    foreach(name memory_access backup_detect idle fusion)
        add_executable(gba_${name}_test tests/${name}_test.c)
        target_link_libraries(gba_${name}_test PRIVATE emu_core)
        target_compile_options(gba_${name}_test PRIVATE -UNDEBUG)
        add_test(NAME gba_${name} COMMAND gba_${name}_test "${CMAKE_CURRENT_BINARY_DIR}/test.rom")
    endforeach()
endif()

if(EMU_NATIVE_TEST)
    add_executable(gba_gameplay_probe tests/gameplay_probe.c)
    target_link_libraries(gba_gameplay_probe PRIVATE emu_core)
    target_compile_options(gba_gameplay_probe PRIVATE -UNDEBUG)
endif()
