cmake_minimum_required(VERSION 3.20)
project(pxa_navigation_ime LANGUAGES C CXX)
list(APPEND CMAKE_MODULE_PATH "${PXA_CMAKE_MODULE_DIR}")
include(PxaGuest)
pxa_add_component(pxa_main COMPONENT_ID main NO_ALLOCATOR SOURCES main.c)
pxa_add_component(pxa_provider COMPONENT_ID provider NO_ALLOCATOR SOURCES provider.cpp)
target_compile_features(pxa_provider PRIVATE cxx_std_20)
target_include_directories(pxa_provider PRIVATE "${PXA_CPP_SDK_DIR}/include")
foreach(target pxa_main pxa_provider)
 target_compile_options(${target} PRIVATE -Oz -flto -fbuiltin)
 target_link_options(${target} PRIVATE -flto -Wl,--lto-O3 -Wl,-z,stack-size=16384)
endforeach()
target_compile_options(pxa_provider PRIVATE -fno-exceptions -fno-rtti)
