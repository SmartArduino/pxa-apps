cmake_minimum_required(VERSION 3.20)
project(pxa_mindustry LANGUAGES C CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../common/cpp/Application.cmake")
pxa_add_callback_component(pxa_main COMPONENT_ID main SOURCES main.cpp core.cpp systems.cpp construction.cpp production.cpp power.cpp player.cpp save.cpp runtime.cpp renderer.cpp renderer_legacy.cpp)
target_compile_features(pxa_main PRIVATE cxx_std_17)
target_compile_definitions(pxa_main PRIVATE PXA_POCKET_RESPONSIVE=1 PXA_POCKET_FILL=1 PXA_POCKET_SYSTEM_TEXT=1)
target_compile_options(pxa_main PRIVATE -flto)
target_link_options(pxa_main PRIVATE -flto -Wl,-z,stack-size=131072)

option(MINDUSTRY_LEGACY_RENDERER "Build the old software/Canvas reference" OFF)
option(MINDUSTRY_PROFILE "Log bounded rendering/cache telemetry" OFF)
if(MINDUSTRY_LEGACY_RENDERER)
 target_compile_definitions(pxa_main PRIVATE MINDUSTRY_LEGACY_RENDERER=1)
endif()
if(MINDUSTRY_PROFILE)
 target_compile_definitions(pxa_main PRIVATE MINDUSTRY_PROFILE=1)
endif()
