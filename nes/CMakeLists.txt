file(GLOB PACKAGE_ROMS CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/assets/roms"
    "assets/roms/*.nes" "assets/roms/*.NES")
set(PACKAGE_ROM_HEADER "static const char *const package_roms[] = {\n")
foreach(rom IN LISTS PACKAGE_ROMS)
    string(REPLACE "\\" "\\\\" escaped "${rom}")
    string(REPLACE "\"" "\\\"" escaped "${escaped}")
    string(APPEND PACKAGE_ROM_HEADER "\"${escaped}\",\n")
endforeach()
string(APPEND PACKAGE_ROM_HEADER "0};\n")
# Keep the generated header stable across target reconfiguration, so packaging
# x86 and Xtensa does not run the whole LTO link twice for identical sources.
set(OLD_PACKAGE_ROM_HEADER "")
if(EXISTS "${CMAKE_CURRENT_BINARY_DIR}/package_roms.h")
    file(READ "${CMAKE_CURRENT_BINARY_DIR}/package_roms.h" OLD_PACKAGE_ROM_HEADER)
endif()
if(NOT PACKAGE_ROM_HEADER STREQUAL OLD_PACKAGE_ROM_HEADER)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/package_roms.h" "${PACKAGE_ROM_HEADER}")
endif()
file(GLOB NES_MAPPERS CONFIGURE_DEPENDS vendor/nofrendo/mappers/*.c)
