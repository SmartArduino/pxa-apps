cmake_minimum_required(VERSION 3.20)
project(pxa_navigation LANGUAGES CXX)
set(PXA_COROUTINE_SLOT_BYTES 8192)
set(PXA_COROUTINE_SLOT_COUNT 12)
list(APPEND CMAKE_MODULE_PATH "${PXA_CMAKE_MODULE_DIR}")
include(PxaGuest)
pxa_add_app(pxa_main COMPONENT_ID main SOURCES main.cpp)
target_compile_options(pxa_main PRIVATE -Oz -flto -fbuiltin)
set(PXA_WIFI_LOCATION_SDK "${CMAKE_CURRENT_SOURCE_DIR}/../../../extensions/wifi-location/sdk" CACHE PATH "Wi-Fi location Guest SDK")
target_include_directories(pxa_main PRIVATE "${PXA_WIFI_LOCATION_SDK}")
target_link_options(pxa_main PRIVATE -flto -Wl,--lto-O3 -Wl,-z,stack-size=65536)
# Test builds may target a local API fixture; release builds always use HTTPS.
if(DEFINED ENV{PXA_NAV_TEST_ORIGIN})
    target_compile_definitions(pxa_main PRIVATE PXA_NAV_API_ORIGIN="$ENV{PXA_NAV_TEST_ORIGIN}" PXA_NAV_FIXTURE=1)
endif()
