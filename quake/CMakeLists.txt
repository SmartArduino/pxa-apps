cmake_minimum_required(VERSION 3.20)
project(pxa_quake C CXX)
# The signed package declares Core 2.1 main-loop execution and native PCM.
if(NOT DEFINED PXA_RECROF_MAIN_LOOP)
    set(PXA_RECROF_MAIN_LOOP ON)
endif()
if(NOT DEFINED PXA_RECROF_NATIVE_PCM)
    set(PXA_RECROF_NATIVE_PCM ON)
endif()
option(PXA_RECROF_HOST_BLIT "Resolve indexed output in GameRender" ON)
option(PXA_RECROF_BULK_READ_COPY "Use bulk copies for large resource reads" ON)
option(PXA_RECROF_PACKED_GAME_CODEC "Use layout-checked command encoding" ON)
option(PXA_QUAKE_LIT_CACHE "Reuse exact light interpolation coefficients" ON)
option(PXA_QUAKE_LIGHTMAP_CACHE "Cache unchanged packed surface light grids" ON)
set(PXA_QUAKE_FRAME_PROFILE ON)
set(PXA_RECROF_ENGINE_STACK_BYTES 98304)
set(PXA_RECROF_CALLBACK_STACK_BYTES 16384)
option(PXA_QUAKE_LIT_CACHE "Enable experimental exact-pixel lighting cache" OFF)
option(PXA_QUAKE_LIGHTMAP_CACHE "Cache unchanged packed surface light grids" OFF)
option(PXA_QUAKE_STATIC_GLOBALS "Use direct addresses for the fixed engine state" ON)
option(PXA_QUAKE_RENDER_PROBE "Private same-AOT render comparisons and sampled surface timings" OFF)
option(PXA_QUAKE_HOST_AUDIO_MIX "Experimental Audio 0.11 synchronous PCM accumulation" OFF)
set(PXA_QUAKE_COLOR_CACHE OFF CACHE BOOL "Exact static shaded tiles remain experimental" FORCE)
include(../common/recrof/Engine.cmake)
file(GLOB ENGINE CONFIGURE_DEPENDS vendor/quake/*.c)
# Keep the renderer and movement math fast without expanding cold parsers,
# menu logic and the generated QuakeC program in the AOT instruction cache.
foreach(source IN LISTS ENGINE)
    get_filename_component(name "${source}" NAME)
    if(NOT name MATCHES "^(d_|r_)" AND NOT name MATCHES "^(draw|mathlib|world)\\.c$")
        set_source_files_properties("${source}" PROPERTIES COMPILE_OPTIONS -Os)
    endif()
endforeach()
pxa_recrof_component(TITLE "Quake" SOURCES ${ENGINE} port.c extmemory.c port/callbacks.c ../common/recrof/vflash.c
    INCLUDES port vendor/quake DEFINITIONS ESP_PLATFORM=1 PORT_PACED_BY_MAIN=1)
if(PXA_QUAKE_FRAME_PROFILE)
    target_compile_definitions(pxa_main PRIVATE PXA_FRAME_PROFILE=1)
endif()
if(PXA_QUAKE_HOST_AUDIO_MIX)
    if(NOT PXA_RECROF_NATIVE_PCM)
        message(FATAL_ERROR "Host audio mix comparisons require the original-rate PCM path")
    endif()
    target_compile_definitions(pxa_main PRIVATE PXA_QUAKE_HOST_AUDIO_MIX=1)
endif()
if(PXA_QUAKE_RENDER_PROBE)
    if(NOT PXA_QUAKE_FRAME_PROFILE OR NOT PXA_RECROF_HOST_BLIT)
        message(FATAL_ERROR "Render probe requires frame profiling and native blit")
    endif()
    target_compile_definitions(pxa_main PRIVATE PXA_QUAKE_RENDER_PROBE=1)
endif()

# Enable only for measured comparisons until the GameKit A/B is complete.
if(PXA_QUAKE_LIT_CACHE)
    target_compile_definitions(pxa_main PRIVATE PXA_LIT_ROW_CACHE=1)
endif()
if(PXA_QUAKE_LIGHTMAP_CACHE)
    target_sources(pxa_main PRIVATE port/lightmap_cache.c)
    target_compile_definitions(pxa_main PRIVATE PXA_QUAKE_LIGHTMAP_CACHE=1)
endif()
if(PXA_QUAKE_COLOR_CACHE)
    if(NOT PXA_QUAKE_LIGHTMAP_CACHE)
        message(FATAL_ERROR "Exact color tiles require light-grid generations")
    endif()
    target_sources(pxa_main PRIVATE port/color_cache.c)
    target_compile_definitions(pxa_main PRIVATE PXA_QUAKE_COLOR_CACHE=1)
endif()
if(PXA_QUAKE_STATIC_GLOBALS)
    target_compile_definitions(pxa_main PRIVATE PXA_STATIC_QUAKE_GLOBALS=1)
endif()


target_compile_definitions(pxa_main PRIVATE PORT_ENGINE_CONTROLLER=1 PORT_EXTERNAL_GAME_DATA=1 PORT_COOPERATE_BUDGET_US=32000
    PORT_TICK_US=8000 PORT_CONTINUATION_BYTES=16384
    PORT_VFLASH_CACHE_PAGES=128 PORT_VFLASH_COOPERATE=1 PORT_VFLASH_PAGED_NVM=1)
