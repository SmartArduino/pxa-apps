cmake_minimum_required(VERSION 3.20)
project(pxa_game_boy C)
set(GB_ENGINE vendor/gnuboy/gnuboy.c vendor/gnuboy/cpu.c vendor/gnuboy/hw.c
    vendor/gnuboy/lcd.c vendor/gnuboy/sound.c)
set(GB_CORE core.c gb_profile.c ../common/rom/rom_cache.c ${GB_ENGINE})
set(GB_UI_SOURCES controls.c font8x8.c ui_font.c)
set(GB_DEFINITIONS)
option(GB_TILE_LUT "Experimental nibble table; disabled after S31 regression" OFF)
option(GB_HALT_BATCH "Coalesce HALT iterations up to the original event batch" ON)
option(GB_SPRITE_SORT "Stable insertion sort for at most ten sprites" ON)
option(GB_WAIT_BATCH "Experimental RAM flag loop batching" OFF)
foreach(feature TILE_LUT HALT_BATCH SPRITE_SORT WAIT_BATCH)
    if(GB_${feature})
        list(APPEND GB_DEFINITIONS GNUBOY_${feature}=1)
    endif()
endforeach()
if("$ENV{GB_PROFILE_PHASES}" STREQUAL "1")
    list(APPEND GB_DEFINITIONS PXA_GB_PROFILE_PHASES=1)
endif()
if(GB_NATIVE_TEST)
    enable_testing()
    add_compile_options(-UNDEBUG)
    add_library(gb_core STATIC ${GB_CORE})
    target_include_directories(gb_core PUBLIC . vendor/gnuboy)
    target_compile_definitions(gb_core PUBLIC ${GB_DEFINITIONS})
    target_compile_options(gb_core PRIVATE -O3 -fno-strict-aliasing)
    target_link_libraries(gb_core PUBLIC m)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/test.gb" "${CMAKE_CURRENT_BINARY_DIR}/test.gbc"
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/make_test_rom.py" "${CMAKE_CURRENT_BINARY_DIR}/test.gb"
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/make_test_rom.py" "${CMAKE_CURRENT_BINARY_DIR}/test.gbc" --cgb
        DEPENDS tests/make_test_rom.py)
    add_custom_target(gb_test_roms ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/test.gb" "${CMAKE_CURRENT_BINARY_DIR}/test.gbc")
    foreach(name core dump_video save benchmark)
        if(name STREQUAL "benchmark")
            set(source tests/benchmark.c)
            set(executable gb_benchmark)
        else()
            set(source tests/${name}_test.c)
            if(name STREQUAL "dump_video")
                set(source tests/dump_video.c)
            endif()
            set(executable gb_${name}_test)
        endif()
        add_executable(${executable} ${source})
        target_link_libraries(${executable} PRIVATE gb_core)
    endforeach()
    add_test(NAME gb_core_dmg COMMAND gb_core_test "${CMAKE_CURRENT_BINARY_DIR}/test.gb" 120)
    add_test(NAME gb_core_cgb COMMAND gb_core_test "${CMAKE_CURRENT_BINARY_DIR}/test.gbc" 120)
    if(GB_WAIT_BATCH OR GB_HALT_BATCH)
        add_executable(gb_wait_test tests/wait_test.c)
        target_link_libraries(gb_wait_test PRIVATE gb_core)
        add_test(NAME gb_wait COMMAND gb_wait_test "${CMAKE_CURRENT_BINARY_DIR}/test.gb")
    endif()
    add_executable(gb_paging_test tests/paging_test.c)
    target_link_libraries(gb_paging_test PRIVATE gb_core)
    add_test(NAME gb_paging COMMAND gb_paging_test "${CMAKE_CURRENT_BINARY_DIR}/test.gb")
    add_test(NAME gb_save COMMAND gb_save_test "${CMAKE_CURRENT_BINARY_DIR}/test.gb")
    add_executable(gb_pacer_test tests/pacer_test.c)
    target_include_directories(gb_pacer_test PRIVATE .)
    add_test(NAME gb_pacer COMMAND gb_pacer_test)
    add_executable(gb_audio_test tests/audio_test.c audio.c)
    target_include_directories(gb_audio_test PRIVATE .)
    target_link_libraries(gb_audio_test PRIVATE m)
    add_test(NAME gb_audio COMMAND gb_audio_test)
    add_executable(gb_video_test tests/video_test.c video_renderer.c)
    target_include_directories(gb_video_test PRIVATE .)
    add_test(NAME gb_video COMMAND gb_video_test)
    foreach(name render_cache controls timing)
        if(name STREQUAL "timing")
            add_executable(gb_${name}_test tests/${name}_test.c)
        else()
            add_executable(gb_${name}_test tests/${name}_test.c controls.c font8x8.c video_renderer.c)
        endif()
        target_include_directories(gb_${name}_test PRIVATE .)
        add_test(NAME gb_${name} COMMAND gb_${name}_test)
    endforeach()
    if(GB_GUEST_SDK_INCLUDE)
        list(APPEND CMAKE_MODULE_PATH "${GB_GUEST_SDK_INCLUDE}/../../cmake")
        get_filename_component(PXA_GUEST_SDK_DIR "${GB_GUEST_SDK_INCLUDE}/.." ABSOLUTE)
        set(PXA_ARTIFACT_DIR "${CMAKE_CURRENT_BINARY_DIR}/artifacts")
        include(PxaGuest)
        add_executable(gb_frontend_test tests/frontend_test.c audio.c video_renderer.c ${GB_UI_SOURCES})
        target_link_libraries(gb_frontend_test PRIVATE gb_core)
        pxa_add_translations(gb_frontend_test SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/i18n/messages.yaml" LANGUAGE c)
        target_include_directories(gb_frontend_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${GB_GUEST_SDK_INCLUDE}")
        target_compile_options(gb_frontend_test PRIVATE -Wno-attributes)
        add_test(NAME gb_frontend COMMAND gb_frontend_test "${CMAKE_CURRENT_BINARY_DIR}/test.gb")
    endif()
else()
    list(APPEND CMAKE_MODULE_PATH "${PXA_CMAKE_MODULE_DIR}")
    include(PxaGuest)
    pxa_add_component(pxa_main COMPONENT_ID main SOURCES main.c audio.c video_renderer.c ${GB_UI_SOURCES} ${GB_CORE})
    target_include_directories(pxa_main PRIVATE . vendor/gnuboy "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_options(pxa_main PRIVATE -flto -O3 -fno-strict-aliasing)
    target_compile_definitions(pxa_main PRIVATE ${GB_DEFINITIONS} PXA_GUEST)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/package.json" GB_PACKAGE_JSON)
    string(JSON GB_PACKAGE_VERSION GET "${GB_PACKAGE_JSON}" version)
    target_compile_definitions(pxa_main PRIVATE GB_APP_VERSION="${GB_PACKAGE_VERSION}")
    string(JSON GB_MEMORY_BYTES GET "${GB_PACKAGE_JSON}" build linear_memory maximum_bytes)
    target_link_options(pxa_main PRIVATE -flto "-Wl,--initial-memory=${GB_MEMORY_BYTES}")
endif()
