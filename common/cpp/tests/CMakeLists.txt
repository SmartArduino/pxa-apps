cmake_minimum_required(VERSION 3.25)
project(pxa_cpp_games_native C CXX)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(APP_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(PXA_SYSTEM_DIR "${APP_ROOT}/../../deps/pxa-system" CACHE PATH "PXA system source checkout")
if(NOT EXISTS "${PXA_SYSTEM_DIR}/sdk/guest-cpp/src/runtime.cpp")
  message(FATAL_ERROR "Set PXA_SYSTEM_DIR to the pxa-system checkout")
endif()
option(SANITIZE "Address and undefined behavior sanitizers" OFF)
if(SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_compile_options(-UNDEBUG -Wall -Wextra -Wno-attributes -Wno-missing-field-initializers)
add_library(sdk STATIC)
foreach(unit runtime game net ipc work surface)
 target_sources(sdk PRIVATE "${PXA_SYSTEM_DIR}/sdk/guest-cpp/src/${unit}.cpp")
endforeach()
target_include_directories(sdk PUBLIC "${PXA_SYSTEM_DIR}/sdk/guest-cpp/include")
target_compile_options(sdk PRIVATE -fno-exceptions -fno-rtti)
add_library(raster STATIC "${PXA_SYSTEM_DIR}/libpxa/src/services/surface/raster.c" "${PXA_SYSTEM_DIR}/libpxa/src/core/wire.c")
target_include_directories(raster PUBLIC "${PXA_SYSTEM_DIR}/libpxa/include")
set(JUMP "${APP_ROOT}/jump-jump-3d-cpp")
set(PIXEL "${APP_ROOT}/pixel-dungeon-cpp")
add_library(jump_core STATIC "${JUMP}/jump3d_game.cpp")
target_include_directories(jump_core PUBLIC "${JUMP}")
add_library(jump_render STATIC "${JUMP}/jump3d_render.cpp" "${JUMP}/jump3d_palette.cpp" "${JUMP}/jump3d_font.cpp")
target_compile_definitions(jump_render PUBLIC J3_SKIP_PROBE=1)
target_link_libraries(jump_render PUBLIC sdk jump_core)
add_executable(jump_selftest "${JUMP}/tools/selftest.cpp")
target_link_libraries(jump_selftest PRIVATE jump_core)
add_executable(jump_preview "${JUMP}/tools/preview.cpp")
target_link_libraries(jump_preview PRIVATE jump_render raster)
add_library(pixel_core STATIC)
foreach(unit game dungeon layout input font assets strings font_data)
 target_sources(pixel_core PRIVATE "${PIXEL}/${unit}.cpp")
endforeach()
target_include_directories(pixel_core PUBLIC "${PIXEL}")
target_compile_definitions(pixel_core PUBLIC PD_EMBED_ASSETS=1)
target_link_libraries(pixel_core PUBLIC sdk)
add_executable(pixel_selftest "${PIXEL}/tools/selftest.cpp")
target_link_libraries(pixel_selftest PRIVATE pixel_core)
add_executable(pixel_save_fixture reference/pixel_save_fixture.cpp)
target_link_libraries(pixel_save_fixture PRIVATE pixel_core)
add_executable(pixel_layout_probe "${PIXEL}/tools/layout_probe.cpp")
target_link_libraries(pixel_layout_probe PRIVATE pixel_core)
add_executable(pixel_render_test "${PIXEL}/tools/render_test.cpp" "${PIXEL}/render.cpp")
target_link_libraries(pixel_render_test PRIVATE pixel_core raster)
add_executable(game_arcade_test "${PXA_SYSTEM_DIR}/sdk/guest-cpp/tests/game_arcade_test.cpp")
target_link_libraries(game_arcade_test PRIVATE sdk raster)
enable_testing()
add_test(NAME jump_game COMMAND jump_selftest)
add_test(NAME pixel_game COMMAND pixel_selftest)
add_test(NAME pixel_raster COMMAND pixel_render_test)
add_test(NAME sdk_arcade COMMAND game_arcade_test)
add_test(NAME jump_raster COMMAND jump_preview --frames 3 --auto 8 --out "${CMAKE_BINARY_DIR}/jump")

# Optional, reproducible reference comparison; the maintained games use only
# the C++ SDK. This target deliberately retains the original C SDK baseline.
option(PXA_C_REFERENCE_BENCH "Build the original C rendering comparison" OFF)
if(PXA_C_REFERENCE_BENCH)
 add_library(pixel_reference STATIC reference/pixel_c_driver.c)
 foreach(unit game dungeon font_data font assets strings render layout input)
  target_sources(pixel_reference PRIVATE "${APP_ROOT}/pixel-dungeon/${unit}.c")
 endforeach()
 target_include_directories(pixel_reference PRIVATE "${APP_ROOT}/pixel-dungeon" "${PXA_SYSTEM_DIR}/sdk/guest-c/include")
 add_executable(pixel_reference_bench reference/pixel_bench.cpp "${PIXEL}/render.cpp")
 target_link_libraries(pixel_reference_bench PRIVATE pixel_core pixel_reference raster)
 add_test(NAME pixel_reference_equivalence COMMAND pixel_reference_bench)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/reference/prepare_jump.py"
  "${APP_ROOT}/jump-jump-3d/tools/preview_render.c" "${CMAKE_CURRENT_BINARY_DIR}/jump_c_preview.c" COMMAND_ERROR_IS_FATAL ANY)
 add_executable(jump_reference_preview "${CMAKE_CURRENT_BINARY_DIR}/jump_c_preview.c")
 foreach(unit jump3d_game jump3d_render jump3d_palette jump3d_font)
  target_sources(jump_reference_preview PRIVATE "${APP_ROOT}/jump-jump-3d/${unit}.c")
 endforeach()
 target_compile_definitions(jump_reference_preview PRIVATE J3_SKIP_PROBE=1)
 target_include_directories(jump_reference_preview PRIVATE "${APP_ROOT}/jump-jump-3d" "${PXA_SYSTEM_DIR}/sdk/guest-c/include")
 target_link_libraries(jump_reference_preview PRIVATE raster m)
 add_test(NAME jump_reference COMMAND jump_reference_preview --frames 3 --auto 4 --out "${CMAKE_BINARY_DIR}/jump-reference")
endif()
